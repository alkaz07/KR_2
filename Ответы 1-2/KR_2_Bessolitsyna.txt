public class Test2{

    {
        a.inc(); // передает счетчик вызовов метода (начинается с 0)
        if(x<100) recFunc(x*3+1, a);// цикл идет пока х меньше 100
        System.out.println(x); //печатает получившийся х
    }

    public static void main (String[] args)
    {
        A o = new A(); // создаем объект а

            recFunc(i, o);

class A{
    private int counter;
    public A() {counter = 0;}/// счетчик
    public void inc() {counter++;} /// увеличиваем счетчик на 1
    public String toString() //печатает счетчик
    { return "A("+counter+")";}
}
