class Oreshek {
    private final double weight;

    public Oreshek() {
        this.weight = 12.5;
    }

    public double getWeight() {
        return weight;
    }
}

class Tree {
    public Oreshek[] growArrayOfNuts(int count) {
        Oreshek[] nuts = new Oreshek[count];
        for (int i = 0; i < count; i++) {
            nuts[i] = new Oreshek();
        }
        return nuts;
    }
}

class Belochka {
    public double collectNuts(Oreshek[] nuts) {
        double totalWeight = 0;
        for (Oreshek nut : nuts) {
            System.out.println("ура, еще орех!");
            totalWeight += nut.getWeight();
        }
        System.out.println("Общий вес собранных орехов: " + totalWeight + " г");
        return totalWeight;
    }
}
public class Main {
    public static void main(String[] args) {
        Tree tree = new Tree();
        Belochka belochka = new Belochka();

        Oreshek[] nuts = tree.growArrayOfNuts(15);
        belochka.collectNuts(nuts);
    }
}
